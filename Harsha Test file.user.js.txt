        try {
            const stored = localStorage.getItem('tm_custom_' + key);
            return stored ? JSON.parse(stored) : [];
        } catch (e) { return []; }

        localStorage.setItem('tm_custom_' + key, JSON.stringify(options));
    }

    function removeCustomOption(key, value) {

        customs = customs.filter(o => o !== value);
        saveCustomOptions(key, customs);

        addInput.addEventListener('keydown', e => { if (e.key === 'Enter') { e.preventDefault(); addHVA(); } });
        setTimeout(() => searchBox.focus(), 100);
    }

    function renderSubIntentStep(hva) {
        const builtIn = HVA_MAP[hva] || [];
        const custom = getCustomOptions('si_' + hva);
        const all = [...builtIn, ...custom];

        let html = `<h3>📋 Offline Sub Intent — Step 2</h3>`;
        html += `<div class="tm-step-label">Step 2 of 2 — Select Sub Intent</div>`;
        html += `<div class="tm-hva-badge">📁 ${hva}</div>`;
        html += `<input type="text" class="tm-search" placeholder="🔍 Search sub-intents..." />`;

        builtIn.forEach(si => {
            html += `<div class="tm-opt-row" data-search="${si.toLowerCase()}">`;
            html += `<button class="tm-opt tm-si-opt" data-val="${si}">${si}</button></div>`;
        });

        if (custom.length > 0) {
            html += `<div class="tm-custom-label">⭐ Custom Sub Intents</div>`;
            custom.forEach(si => {
                html += `<div class="tm-opt-row" data-search="${si.toLowerCase()}">`;
                html += `<button class="tm-opt tm-opt-custom tm-si-opt" data-val="${si}">⭐ ${si}</button>`;
                html += `<button class="tm-del-btn tm-del-si" data-del="${si}">✕</button></div>`;
            });
        }

        html += `<div class="tm-add-section"><div class="tm-add-label">➕ Add Custom Sub Intent for "${hva}"</div>`;
        html += `<div class="tm-add-row"><input type="text" class="tm-add-input tm-add-si" placeholder="Type new sub-intent..." />`;
        html += `<button class="tm-add-btn tm-add-si-btn">ADD</button></div></div>`;
        html += `<button class="tm-back">⬅ Back to HVA List</button>`;
        html += `<button class="tm-cancel">✖ Cancel</button>`;
        subIntentPicker.innerHTML = html;

        const searchBox = subIntentPicker.querySelector('.tm-search');
        searchBox.addEventListener('input', function () {
            const q = this.value.toLowerCase();
            subIntentPicker.querySelectorAll('.tm-opt-row').forEach(r =>
                r.classList.toggle('tm-hide', !r.dataset.search.includes(q)));
        });

        subIntentPicker.querySelector('.tm-back').addEventListener('click', renderHVAStep);
        subIntentPicker.querySelector('.tm-cancel').addEventListener('click', hideAllPickers);

        subIntentPicker.querySelectorAll('.tm-si-opt').forEach(btn => {
            btn.addEventListener('click', function () {
                const val = this.dataset.val;
                debugLog('Sub-intent selected: ' + val);
                hideAllPickers();
                setTimeout(() => {
                    const field = findSubIntentField();
                    if (field) { smartFill(field, val); showToast(`✅ Sub Intent: ${val}`); }
                    else alert('❌ Cannot find sub-intent field!');
                }, 500);
            });
        });

        subIntentPicker.querySelectorAll('.tm-del-si').forEach(btn => {
            btn.addEventListener('click', function (e) {
                e.stopPropagation();
                const val = this.dataset.del;
                showDeleteConfirm(val, () => {
                    removeCustomOption('si_' + hva, val);
                    showToast(`🗑️ Deleted: ${val}`);
                    renderSubIntentStep(hva);
                });
            });
        });

        const addInput = subIntentPicker.querySelector('.tm-add-si');
        const addBtn = subIntentPicker.querySelector('.tm-add-si-btn');
        function addSI() {
            const v = addInput.value.trim();
            if (!v) return;
            if (all.includes(v)) { showToast('⚠️ Already exists!'); return; }
            const c = getCustomOptions('si_' + hva); c.push(v); saveCustomOptions('si_' + hva, c);
            showToast(`✅ Added: ${v}`);
            renderSubIntentStep(hva);
        }
        addBtn.addEventListener('click', addSI);
        addInput.addEventListener('keydown', e => { if (e.key === 'Enter') { e.preventDefault(); addSI(); } });
        setTimeout(() => searchBox.focus(), 100);
    }

    function showSubIntentPicker() {
        overlay.style.display = 'block';
        subIntentPicker.style.display = 'block';

    }

    // ═══════════════════════════════════════════════
    //  FIND FIELDS
    // ═══════════════════════════════════════════════

    function findSubIntentField() {
        const selectors = [
            'input[placeholder*="offline sub intent" i]',
            'input[placeholder*="correct offline" i]',
            'input[placeholder*="sub intent" i]'
        ];
        for (const sel of selectors) {
            try {
                const inp = document.querySelector(sel);
                if (inp) return inp;
            } catch(e) {}
        }

        const allEls = document.querySelectorAll('label, span, p, div, h3, h4, h5, h6');
        for (const el of allEls) {
            if (/accurate offline sub intent/i.test(el.textContent.trim())) {
                let parent = el.parentElement;
                for (let i = 0; i < 6; i++) {
                    if (!parent) break;
                    const inp = parent.querySelector('input[type="text"], input:not([type])');
                    if (inp) return inp;
                    parent = parent.parentElement;
                }
            }
        }

        const allInputs = document.querySelectorAll('input[type="text"], input:not([type])');
        for (const inp of allInputs) {
            const ph = (inp.placeholder || '').toLowerCase();
            if (ph.includes('offline') || ph.includes('sub intent')) return inp;
        }
        return null;
    }

    function findFieldByConfig(config) {
        try {
            let inp = document.querySelector(`input[placeholder*="${config.fieldPlaceholder}" i]`);
            if (inp) return inp;
        } catch(e) {}

        const allEls = document.querySelectorAll('label, span, p, div, h3, h4, h5');
        for (const el of allEls) {
            const t = el.textContent.trim();
            if (t === config.fieldLabel || t === config.fieldLabel + ' *') {
                let parent = el.parentElement;
                for (let i = 0; i < 5; i++) {
                    if (!parent) break;
                    const inp = parent.querySelector('input');
                    if (inp) return inp;
                    parent = parent.parentElement;
                }
            }
        }
        return null;
    }

    // ═══════════════════════════════════════════════
    //  SMART FILL (React-compatible)
    // ═══════════════════════════════════════════════

    function smartFill(input, value) {
        debugLog('Filling → ' + value);
        input.focus();
        input.click();
        setTimeout(() => {
            const nativeSet = Object.getOwnPropertyDescriptor(
                window.HTMLInputElement.prototype, 'value'
            ).set;
            nativeSet.call(input, '');
            input.dispatchEvent(new Event('input', { bubbles: true }));
            input.dispatchEvent(new Event('change', { bubbles: true }));
            setTimeout(() => {
                nativeSet.call(input, value);
                input.setAttribute('value', value);
                input.dispatchEvent(new Event('input', { bubbles: true }));
                input.dispatchEvent(new Event('change', { bubbles: true }));
                setTimeout(() => {
                    input.dispatchEvent(new Event('blur', { bubbles: true }));
                    debugLog('✅ Filled: ' + input.value);
                }, 200);
            }, 100);
        }, 100);
    }

    // ═══════════════════════════════════════════════
    //  🔥 V15: TARGETED "No" RADIO BINDING
    //  ─────────────────────────────────────────────
    //  Instead of listening to ALL clicks on the page,
    //  we FIND the specific "No" radio buttons for each
    //  question and attach listeners ONLY to those.
    //
    //  Uses MutationObserver + periodic scan because
    //  React re-renders the DOM dynamically.
    // ═══════════════════════════════════════════════

    // Track which radios we've already bound
    const boundRadios = new WeakSet();

    // Debounce
    let lastTriggerTime = 0;
    const DEBOUNCE_MS = 2000;

    /**
     * Find the question container element by searching
     * for its label text on the page.
     */
    function findQuestionContainer(labelTexts) {
        const allEls = document.querySelectorAll(
            'span, label, div, p, h1, h2, h3, h4, h5, h6, legend'
        );

        for (const el of allEls) {
            const text = el.textContent.trim().toLowerCase();
            // Element text should be SHORT (it's a label, not a huge container)
            if (text.length > 200) continue;

            for (const lt of labelTexts) {
                if (text.includes(lt)) {
                    debugLog(`Found label: "${el.textContent.trim().substring(0, 50)}" for "${lt}"`);
                    return el;
                }
            }
        }
        return null;
    }

    /**
     * From a question label element, walk UP to find the
     * container that holds the radio buttons, then find
     * the "No" radio inside it.
     */
    function findNoRadioForQuestion(labelEl) {
        let container = labelEl;

        for (let i = 0; i < 10; i++) {
            if (!container) return null;

            // Look for radio buttons in this container
            const radios = container.querySelectorAll('input[type="radio"]');

            if (radios.length >= 2) {
                // Found the radio group — now find the "No" one
                for (const radio of radios) {
                    if (radio.value === 'No') {
                        debugLog(`✅ Found "No" radio: id="${radio.id}" name="${radio.name}"`);
                        return radio;
                    }
                }

                // No radio with value="No" — check by label text
                for (const radio of radios) {
                    const label = radio.closest('label');
                    if (label) {
                        const labelText = label.textContent.trim().toLowerCase();
                        if (labelText === 'no') {
                            debugLog(`✅ Found "No" radio by label text: id="${radio.id}"`);
                            return radio;
                        }
                    }
                }

                debugLog('⚠️ Found radio group but no "No" value. Values: ' +
                    [...radios].map(r => r.value).join(', '));
                return null;
            }

            container = container.parentElement;
        }

        debugLog('⚠️ Could not find radio group');
        return null;
    }

    /**
     * Open the correct picker for a question key
     */
    function openPickerForQuestion(questionDef) {
        // Debounce
        const now = Date.now();
        if (now - lastTriggerTime < DEBOUNCE_MS) {
            debugLog('⏳ Debounced');
            return;
        }
        lastTriggerTime = now;

        // Skip if picker already open
        if (overlay.style.display === 'block') {
            debugLog('⏳ Picker already open');
            return;
        }

        debugLog(`🚀 Opening picker: ${questionDef.key}`);

        badge.textContent = `🟢 ${questionDef.key}: "No" clicked`;
        badge.style.background = '#1d8102';
        setTimeout(() => {
            badge.style.background = '#0073bb';
            badge.textContent = '🟢 TM v15 Active';
        }, 3000);

        setTimeout(() => {
            if (questionDef.type === 'simple') {
                showSimplePicker(questionDef.key);
            } else {
                showSubIntentPicker();
            }
        }, 400);
    }

    /**
     * Bind a click/change listener to a specific "No" radio
     */
    function bindNoRadio(radio, questionDef) {
        if (boundRadios.has(radio)) return; // Already bound
        boundRadios.add(radio);

        debugLog(`🔗 Binding listener: "${questionDef.key}" → radio id="${radio.id}"`);

        // Listen for BOTH change and click
        radio.addEventListener('change', function () {
            if (this.checked && this.value === 'No') {
                debugLog(`📻 [change] "${questionDef.key}" No selected`);
                openPickerForQuestion(questionDef);
            }
        });

        radio.addEventListener('click', function () {
            if (this.value === 'No') {
                debugLog(`🖱️ [click] "${questionDef.key}" No clicked`);
                openPickerForQuestion(questionDef);
            }
        });

        // Also listen on the parent label (Cloudscape wraps radios in labels)
        const label = radio.closest('label');
        if (label) {
            label.addEventListener('click', function (e) {
                // Make sure we're not double-firing from the radio itself
                if (e.target === radio) return;
                if (radio.value === 'No') {
                    debugLog(`🏷️ [label click] "${questionDef.key}" No label clicked`);
                    // Small delay to let the radio get checked
                    setTimeout(() => {
                        if (radio.checked) {
                            openPickerForQuestion(questionDef);
                        }
                    }, 50);
                }
            });
        }
    }

    /**
     * MAIN SCAN: Find all 3 question groups and bind their "No" radios
     */
    function scanAndBind() {
        let boundCount = 0;

        for (const questionDef of QUESTION_MAP) {
            const labelEl = findQuestionContainer(questionDef.labelTexts);
            if (!labelEl) continue;

            const noRadio = findNoRadioForQuestion(labelEl);
            if (!noRadio) continue;

            if (!boundRadios.has(noRadio)) {
                bindNoRadio(noRadio, questionDef);
                boundCount++;
            }
        }

        if (boundCount > 0) {
            debugLog(`✅ Bound ${boundCount} new "No" radio(s)`);
            badge.textContent = `🟢 TM v15 — ${boundCount} bound`;
            setTimeout(() => { badge.textContent = '🟢 TM v15 Active'; }, 2000);
        }

        return boundCount;
    }

    // ═══════════════════════════════════════════════
    //  INITIAL SCAN + OBSERVER + PERIODIC RE-SCAN
    // ═══════════════════════════════════════════════

    // Initial scan after page settles
    setTimeout(() => {
        debugLog('🔍 Initial scan...');
        const count = scanAndBind();
        debugLog(`Initial scan: bound ${count} radios`);
    }, 2000);

    // Re-scan when DOM changes (React re-renders)
    let scanTimeout = null;
    const observer = new MutationObserver(() => {
        if (scanTimeout) clearTimeout(scanTimeout);
        scanTimeout = setTimeout(() => {
            scanAndBind();
        }, 1000);
    });

    observer.observe(document.body, {
        childList: true,
        subtree: true
    });

    // Periodic re-scan every 5s as safety net
    setInterval(() => {
        scanAndBind();
    }, 5000);

    // ═══════════════════════════════════════════════
    //  KEYBOARD SHORTCUTS
    // ═══════════════════════════════════════════════

    document.addEventListener('keydown', (e) => {
        if (e.ctrlKey && e.shiftKey && e.key === 'L') { e.preventDefault(); showSimplePicker('llmIntent'); }
        if (e.ctrlKey && e.shiftKey && e.key === 'K') { e.preventDefault(); showSimplePicker('llmWorkflow'); }
        if (e.ctrlKey && e.shiftKey && e.key === 'J') { e.preventDefault(); showSubIntentPicker(); }
        if (e.ctrlKey && e.shiftKey && e.key === 'D') {
            e.preventDefault();
            debugPanel.style.display = debugPanel.style.display === 'none' ? 'block' : 'none';
        }
        // Force re-scan
        if (e.ctrlKey && e.shiftKey && e.key === 'R') {
            e.preventDefault();
            debugLog('🔄 Manual re-scan...');
            const count = scanAndBind();
            showToast(`🔄 Re-scan: ${count} new bindings`);
        }
    });

    debugLog('🚀 TM v15.0 loaded — Targeted radio binding');

})();
